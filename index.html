    const shuffleArray = function(arr) {

        const copy = arr.slice();

        for (let i = copy.length - 1; i > 0; i--) {

            const j = Math.floor(Math.random() * (i + 1));

            const temp = copy[i];

            copy[i] = copy[j];

            copy[j] = temp;

        }

        return copy;

    };

 

    const fetchDeezerJSONP = function(endpoint) {

        return new Promise(function(resolve, reject) {

            const callbackName = 'dz_cb_' + Math.round(100000 * Math.random());

            const script = document.createElement('script');

            const separator = endpoint.includes('?') ? '&' : '?';

            const timeout = setTimeout(function() { cleanup(); reject(new Error('Zeitüberschreitung bei Deezer.')); }, 10000);

            function cleanup() {

                clearTimeout(timeout);

                delete window[callbackName];

                if (script.parentNode) script.parentNode.removeChild(script);

            }

            window[callbackName] = function(data) {

                cleanup();

                if (data && data.error) reject(new Error(data.error.message || 'Deezer API Fehler'));

                else resolve(data);

            };

            

script.src = 'https://' + 'api.' + 'deezer.com' + endpoint + separator + 'output=jsonp&callback=' + callbackName; 

            script.onerror = function() { cleanup(); reject(new Error('Netzwerkfehler.')); };

            document.body.appendChild(script);

        });

    };

 

    let audioCtx = null;

    const unlockAudio = function() {

        try {

            if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();

            if (audioCtx.state === 'suspended') audioCtx.resume();

        } catch (e) {}

    };

 

    const playSound = function(type) {

        try {

            unlockAudio();

            if (!audioCtx) return;

            const osc = audioCtx.createOscillator();

            const gain = audioCtx.createGain();

            osc.connect(gain);

            gain.connect(audioCtx.destination);

            const now = audioCtx.currentTime;

            if (type === 'win') {

                osc.type = 'triangle';

                osc.frequency.setValueAtTime(523.25, now);

                osc.frequency.setValueAtTime(659.25, now + 0.12);

                osc.frequency.setValueAtTime(783.99, now + 0.24);

                gain.gain.setValueAtTime(0.25, now);

                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.55);

                osc.start(now);

                osc.stop(now + 0.55);

            } else {

                osc.type = 'sawtooth';

                osc.frequency.setValueAtTime(150, now);

                osc.frequency.linearRampToValueAtTime(110, now + 0.35);

                gain.gain.setValueAtTime(0.25, now);

                gain.gain.exponentialRampToValueAtTime(0.01, now + 0.4);

                osc.start(now);

                osc.stop(now + 0.4);

            }

        } catch (e) {}

    };

    const extractDeezerPlaylistId = function(input) {

        if (!input) return null;

        const match = input.match(/playlist\/(\d+)/);

        if (match && match[1]) return match[1];

        if (/^\d+$/.test(input.trim())) return input.trim();

        return null;

    };

 

    const loadDeezerPlaylistTracks = async function(playlistId, statusCallback) {

        let allTracks = [];

        let index = 0;

        const limit = 100;

        let hasMore = true;

        while (hasMore && allTracks.length < 1000) {

            statusCallback('Lade Songs aus Deezer-Playlist (' + allTracks.length + ' geladen)...');

            const data = await fetchDeezerJSONP('/playlist/' + playlistId + '/tracks?index=' + index + '&limit=' + limit);

            if (!data || !data.data || data.data.length === 0) hasMore = false;

            else {

                allTracks = allTracks.concat(data.data);

                index += limit;

                if (!data.next) hasMore = false;

            }

        }

        return allTracks.filter(function(t) {

            return t && t.id && t.preview && t.preview.length > 5 && t.album && (t.album.cover_big || t.album.cover_medium);

        });

    };

    const resolveCuratedTrack = async function(pair) {

    const q = pair[0] + ' ' + pair[1];

    try {

        const data = await fetchDeezerJSONP('/search?q=' + encodeURIComponent(q) + '&limit=3');

        if (data && data.data) {

            const valid = data.data.find(function(t) {

                return t && t.id && t.preview && t.preview.length > 5 && t.album && (t.album.cover_big || t.album.cover_medium);

            });

            return valid || null;

        }

    } catch (e) {}

    return null;

};

    const loadCuratedCatalogTracks = async function(catalogKey, statusCallback) {

        const catalog = window.GUERILLAZ_CATALOG || {};

        const entry = catalog[catalogKey];

        if (!entry) throw new Error('Katalog-Playlist nicht gefunden.');

        const needed = Math.min(entry.tracks.length, Math.max(40, gameConfig.rounds * gameConfig.players.length * 2 + 20));

        const sample = shuffleArray(entry.tracks).slice(0, needed);

        const resolved = [];

        const batchSize = 6;

        for (let i = 0; i < sample.length; i += batchSize) {

            statusCallback('Bereite Songs vor (' + resolved.length + ' / ' + needed + ')...');

            const batch = sample.slice(i, i + batchSize);

            const results = await Promise.all(batch.map(resolveCuratedTrack));

            results.forEach(function(r) { if (r) resolved.push(r); });

            if (resolved.length >= gameConfig.rounds * gameConfig.players.length + gameConfig.coverCount + 6) break;

        }

        return resolved;

    };

                currentPlaylistId = 'dz_' + dzId;

                trackPool = await loadDeezerPlaylistTracks(dzId, function(m) { errorEl.innerText = m; });

            }

 

            if (trackPool.length < coverCount) {

                errorEl.innerText = 'Zu wenig abspielbare Songs gefunden (mindestens ' + coverCount + ' benötigt).';

                return;

            }

 

            sessionShownAlbumIds.clear();

            gameState = { currentRound: 1, currentPlayerIndex: 0, isTiebreaker: false, tiebreakerRound: 0, currentCorrectTrack: null, hasAnswered: false };

            errorEl.innerText = '';

            showScreen('game');

            startTurn();

        } catch (e) {

            errorEl.innerText = 'Fehler beim Laden: ' + e.message;

        }

    });

    const buildExpertChoices = async function(correctTrack) {

        const choices = [{ id: correctTrack.id, albumId: correctTrack.album.id, cover: correctTrack.album.cover_big || correctTrack.album.cover_medium, isCorrect: true }];

        const usedAlbums = new Set([correctTrack.album.id]);

        if (correctTrack.artist && correctTrack.artist.id) {

            try {

                const data = await fetchDeezerJSONP('/artist/' + correctTrack.artist.id + '/albums?limit=30');

                if (data && data.data) {

                    const albums = shuffleArray(data.data);

                    for (let i = 0; i < albums.length; i++) {

                        if (choices.length >= gameConfig.coverCount) break;

                        const alb = albums[i];

                        if (alb && alb.id && !usedAlbums.has(alb.id) && (alb.cover_big || alb.cover_medium)) {

                            usedAlbums.add(alb.id);

                            choices.push({ id: -alb.id, albumId: alb.id, cover: alb.cover_big || alb.cover_medium, isCorrect: false });

                        }

                    }

                }

            } catch (e) {}

        }

        if (choices.length < gameConfig.coverCount) {

            const fallback = shuffleArray(trackPool);

            for (let i = 0; i < fallback.length; i++) {

                if (choices.length >= gameConfig.coverCount) break;

                const cand = fallback[i];

                if (!usedAlbums.has(cand.album.id)) {

                    usedAlbums.add(cand.album.id);

                    choices.push({ id: cand.id, albumId: cand.album.id, cover: cand.album.cover_big || cand.album.cover_medium, isCorrect: false });

                }

            }

        }

        return shuffleArray(choices);

    };

 

    const buildNormalChoices = function(correctTrack) {

        const choices = [{ id: correctTrack.id, albumId: correctTrack.album.id, cover: correctTrack.album.cover_big || correctTrack.album.cover_medium, isCorrect: true }];

        const usedAlbumIds = new Set([correctTrack.album.id]);

        const usedArtistKeys = new Set([(correctTrack.artist && correctTrack.artist.name ? correctTrack.artist.name : '').toLowerCase().trim()]);

 

        const unseen = shuffleArray(trackPool.filter(function(t) { return !sessionShownAlbumIds.has(t.album.id); }));

        const seen = shuffleArray(trackPool.filter(function(t) { return sessionShownAlbumIds.has(t.album.id); }));

        const orderedPool = unseen.concat(seen);

 

        for (let i = 0; i < orderedPool.length; i++) {

            if (choices.length >= gameConfig.coverCount) break;

            const cand = orderedPool[i];

            const artistKey = (cand.artist && cand.artist.name ? cand.artist.name : '').toLowerCase().trim();

            if (!usedAlbumIds.has(cand.album.id) && (!artistKey || !usedArtistKeys.has(artistKey))) {

                usedAlbumIds.add(cand.album.id);

                if (artistKey) usedArtistKeys.add(artistKey);

                sessionShownAlbumIds.add(cand.album.id);

                choices.push({ id: cand.id, albumId: cand.album.id, cover: cand.album.cover_big || cand.album.cover_medium, isCorrect: false });

            }

        }

        return shuffleArray(choices);

    };

 

    const startTurn = async function() {

        gameState.hasAnswered = false;

        document.getElementById('feedback-box').style.display = 'none';

        renderScoreboard();

 

        const currentPlayer = gameConfig.players[gameState.currentPlayerIndex];

        document.getElementById('round-indicator').innerText = gameState.isTiebreaker

            ? ('⚡ STECHEN – Runde ' + gameState.tiebreakerRound)

            : ('Runde ' + gameState.currentRound + ' von ' + gameConfig.rounds);

        document.getElementById('turn-indicator').innerText = currentPlayer.name + ' ist dran!';

 

        const playedSet = getPlayedIds(currentPlaylistId);

        let availableTracks = trackPool.filter(function(t) { return !playedSet.has(t.id); });

        if (availableTracks.length === 0) {

            resetPlayedIds(currentPlaylistId);

            availableTracks = trackPool.slice();

        }

 

        const correctTrack = availableTracks[Math.floor(Math.random() * availableTracks.length)];

        savePlayedId(currentPlaylistId, correctTrack.id);

        sessionShownAlbumIds.add(correctTrack.album.id);

        gameState.currentCorrectTrack = correctTrack;

 

        const roundChoices = (gameConfig.mode === 'expert')

            ? await buildExpertChoices(correctTrack)

            : buildNormalChoices(correctTrack);

 

        renderCovers(roundChoices);

        playPreview(correctTrack.preview);

    };

    const renderCovers = function(items) {

        const grid = document.getElementById('covers-grid');

        grid.innerHTML = '';

        items.forEach(function(item) {

            const card = document.createElement('div');

            card.dataset.isCorrect = item.isCorrect ? '1' : '0';

            card.style.cssText = 'cursor: pointer; border-radius: 10px; overflow: hidden; border: 4px solid transparent; transition: transform 0.15s, border-color 0.2s; background: #181818; aspect-ratio: 1 / 1;';

            const img = document.createElement('img');

            img.src = item.cover;

            img.alt = 'Album Cover';

            img.style.cssText = 'width: 100%; height: 100%; object-fit: cover; display: block;';

            card.appendChild(img);

            card.addEventListener('mouseenter', function() { if (!gameState.hasAnswered) card.style.transform = 'scale(1.03)'; });

            card.addEventListener('mouseleave', function() { card.style.transform = 'scale(1)'; });

            card.addEventListener('click', function() { handleGuess(item, card); });

            grid.appendChild(card);

        });

    };

 

    const handleGuess = function(selectedItem, clickedCard) {

        if (gameState.hasAnswered) return;

        gameState.hasAnswered = true;

        const isCorrect = selectedItem.isCorrect;

        const currentPlayer = gameConfig.players[gameState.currentPlayerIndex];

 

        const allCards = document.querySelectorAll('#covers-grid > div');

        allCards.forEach(function(card) {

            card.style.cursor = 'default';

            if (card.dataset.isCorrect === '1') {

                card.style.borderColor = '#1DB954';

                card.style.boxShadow = '0 0 20px rgba(29, 185, 84, 0.7)';

            } else if (card === clickedCard && !isCorrect) {

                card.style.borderColor = '#ff4444';

                card.style.opacity = '0.6';

            } else {

                card.style.opacity = '0.45';

            }

        });

 

        const feedbackTitle = document.getElementById('feedback-title');

        const feedbackSong = document.getElementById('feedback-song');

        const artistName = gameState.currentCorrectTrack.artist ? gameState.currentCorrectTrack.artist.name : 'Unbekannt';

        const albumTitle = gameState.currentCorrectTrack.album ? gameState.currentCorrectTrack.album.title : '';

 

        if (isCorrect) {

            currentPlayer.score += 1;

            playSound('win');

            feedbackTitle.innerHTML = '<span style="color: #1DB954;">✔ Richtig! +1 Punkt für ' + currentPlayer.name + '</span>';

        } else {

            playSound('lose');

            feedbackTitle.innerHTML = '<span style="color: #ff5555;">✘ Leider falsch!</span>';

        }

 

        feedbackSong.innerHTML = 'Gesucht: <strong>' + gameState.currentCorrectTrack.title + '</strong> – ' + artistName + (albumTitle ? ' <span style="color:#999;">(' + albumTitle + ')</span>' : '');

        document.getElementById('feedback-box').style.display = 'block';

        renderScoreboard();

    };

 

    document.getElementById('btn-next-turn').addEventListener('click', function() {

        let nextIndex = gameState.currentPlayerIndex + 1;

        while (nextIndex < gameConfig.players.length && !gameConfig.players[nextIndex].activeInTiebreaker) {

            nextIndex++;

        }

        if (nextIndex < gameConfig.players.length) {

            gameState.currentPlayerIndex = nextIndex;

            startTurn();

            return;

        }

        if (!gameState.isTiebreaker) {

            if (gameState.currentRound < gameConfig.rounds) {

                gameState.currentRound++;

                gameState.currentPlayerIndex = 0;

                startTurn();

            } else {

                evaluateGameEnd();

            }

        } else {

            evaluateGameEnd();

        }

    });

 

    const evaluateGameEnd = function() {

        const maxScore = Math.max.apply(null, gameConfig.players.map(function(p) { return p.score; }));

        const leaders = gameConfig.players.filter(function(p) { return p.score === maxScore; });

        if (leaders.length === 1) {

            audioPlayer.pause();

            showWinnerScreen(leaders[0]);

        } else {

            gameState.isTiebreaker = true;

            gameState.tiebreakerRound += 1;

            gameConfig.players.forEach(function(p) { p.activeInTiebreaker = (p.score === maxScore); });

            gameState.currentPlayerIndex = gameConfig.players.findIndex(function(p) { return p.activeInTiebreaker; });

            startTurn();

        }

    };

 

    const renderScoreboard = function() {

        const board = document.getElementById('scoreboard');

        board.innerHTML = '';

        gameConfig.players.forEach(function(p, idx) {

            const isCurrent = idx === gameState.currentPlayerIndex;

            const badge = document.createElement('div');

            badge.style.cssText = 'padding: 8px 16px; border-radius: 20px; font-size: 14px; ' +

                'background: ' + (isCurrent ? '#1DB954' : '#282828') + '; ' +

                'color: ' + (isCurrent ? '#000' : (p.activeInTiebreaker ? '#fff' : '#666')) + '; ' +

                'font-weight: ' + (isCurrent ? 'bold' : 'normal') + '; ' +

                'border: 1px solid ' + (isCurrent ? '#1DB954' : '#383838') + ';';

            badge.innerText = p.name + ': ' + p.score;

            board.appendChild(badge);

        });

    };

 

    const showWinnerScreen = function(winner) {

        showScreen('winner');

            standings.appendChild(row);

        });

    };
